cmake_minimum_required(VERSION 3.22)
project(IRChoir VERSION 0.3.0)
set(CMAKE_CXX_STANDARD 17)
if(APPLE)
    set(IRCHOIR_FORMATS AU VST3)
else()
    set(IRCHOIR_FORMATS VST3)
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(IRChoir COMPANY_NAME "IR Audio" BUNDLE_ID "audio.ir.choir" PLUGIN_MANUFACTURER_CODE IrAu PLUGIN_CODE IrCh FORMATS ${IRCHOIR_FORMATS} PRODUCT_NAME "IRChoir" PLUGIN_NAME "I.R.Choir" AU_MAIN_TYPE kAudioUnitType_MusicEffect IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(IRChoir PRIVATE Source/Plugin.cpp)
target_compile_definitions(IRChoir PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(IRChoir PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)

option(IRCHOIR_BUILD_TESTS "Build smoke tests" ON)
if(IRCHOIR_BUILD_TESTS)
    enable_testing()
    add_executable(IRChoirDSPTests Tests/dsp_tests.cpp)
    add_test(NAME dsp COMMAND IRChoirDSPTests)
    juce_add_console_app(IRChoirSmoke PRODUCT_NAME "IRChoirSmoke")
    target_sources(IRChoirSmoke PRIVATE Tests/host_smoke.cpp)
    target_compile_definitions(IRChoirSmoke PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1 IRCHOIR_TESTING=1)
    target_link_libraries(IRChoirSmoke PRIVATE juce::juce_audio_utils)
    add_test(NAME processor_editor_smoke COMMAND IRChoirSmoke)
endif()
